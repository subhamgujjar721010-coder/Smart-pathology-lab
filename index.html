Role & Google Sheets Cloud Sync

Google Apps Script Code (Supports GET + POST + JSONP — Copy & Deploy as "Anyone")
const SECRET_KEY = "SPL_SYNC_2026"; function getSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(["id", "updatedAt", "json"]); } return sh; } function upsertRecord(rec) { const sh = getSheet("Patients"); const rows = sh.getDataRange().getValues(); const id = String(rec.id || rec.labId); let idx = -1; for (let i = 1; i < rows.length; i++) if (String(rows[i][0]) === id) { idx = i + 1; break; } const vals = [id, rec.updatedAt || new Date().toISOString(), JSON.stringify(rec)]; if (idx > 0) sh.getRange(idx, 1, 1, 3).setValues([vals]); else sh.appendRow(vals); } function doGet(e) { if ((e.parameter.key || "") !== SECRET_KEY) return ContentService.createTextOutput('{"ok":false}').setMimeType(ContentService.MimeType.JSON); if (e.parameter.save) { try { upsertRecord(JSON.parse(e.parameter.save)); } catch (err) {} } const rows = getSheet("Patients").getDataRange().getValues(); const records = []; for (let i = 1; i < rows.length; i++) if (rows[i][2]) records.push(JSON.parse(rows[i][2])); const out = JSON.stringify({ ok: true, records }); if (e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + "(" + out + ")").setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const body = JSON.parse(e.postData.contents || "{}"); if (body.key !== SECRET_KEY) return ContentService.createTextOutput('{"ok":false}').setMimeType(ContentService.MimeType.JSON); if (body.record) upsertRecord(body.record); return ContentService.createTextOutput('{"ok":true}').setMimeType(ContentService.MimeType.JSON); }
